```typescript
export interface BrowserOpenGuard {
  signal: AbortSignal;
  expectedGeneration: number;
}

export interface BrowserRequestGuard {
  signal: AbortSignal;
  expectedGeneration: number;
  owner: ProfileOwnerIdentity;
}

export type BrowserLocatorSpec =
  | { kind: 'role'; role: string; name?: string; exact: true }
  | { kind: 'text'; text: string; exact: true }
  | { kind: 'label'; text: string; exact: true }
  | { kind: 'css'; selector: string };

export interface NavigationObservation {
  redirectOrigins: string[];
  finalOrigin: string;
}

export interface BrowserProbeSession {
  navigate(target: URL, guard: BrowserRequestGuard): Promise<NavigationObservation>;
  waitFor(locator: BrowserLocatorSpec, guard: BrowserRequestGuard): Promise<'visible'>;
  readVisible(locator: BrowserLocatorSpec, guard: BrowserRequestGuard): Promise<string | null>;
  readAttribute(locator: BrowserLocatorSpec, attribute: string, guard: BrowserRequestGuard): Promise<string | null>;
  close(guard: BrowserRequestGuard): Promise<void>;
}

export class LocalPlaywrightBrowserProvider {
  openBackground(input: BrowserOpenInput, guard: BrowserOpenGuard): Promise<BrowserProbeSession>;
  openOfficialLogin(input: InteractiveLoginOpenInput, guard: BrowserOpenGuard): Promise<BrowserProbeSession>;
}
```

`BrowserOpenInput` 只含 installation/build/config 引用、已确认的 read/auth origins、generation/fence 等固定 metadata；provider 内部解析受管 executable/Profile path。launch 前尚无 owner，因此 open 方法只接收 `BrowserOpenGuard`；attach 成功后返回的 session 固定 owner，此后每次操作必须提交完全匹配的 `BrowserRequestGuard`。`InteractiveLoginOpenInput` 额外要求 paired-local human-action receipt id 与固定 SourceId。`BrowserLocatorSpec` 只允许 02-06 sealed adapter 的静态、长度受限常量；不能从 operation payload、API、MCP、页面内容或模型输出构造。它没有 evaluate/function/JS、任意网络请求、click/fill/submit、Cookie、storageState 或 raw Playwright handle。
